# GUI Button for when crime is inactive
custom_button = {
	name = criminal_rating
	potential = { NOT = { has_any_crime_lvl = yes } }
	trigger = { hidden_trigger = { owned_by = FROM } }
	effect = {  }
	tooltip = "criminal_rating_tt"
}

# GUI Button for when crime is active
custom_button = {
	name = criminal_rating_active
	potential = { has_any_crime_lvl = yes }
	trigger = { hidden_trigger = { owned_by = FROM } }
	effect = {
		# Display cost to reduce crime lvl
		custom_tooltip = "crime_cost_reduction_tt"
		if = {
			limit = {
				owned_by = FROM
				NOT = { has_province_modifier = anticrime_campaign }
				has_any_crime_lvl = yes
			}
			# Discounted from having soldiers
			if = {
				limit = { units_in_province = owner }
				remove_treasury_prov = { variable = halved_crime_cost }
			}
			else = { remove_treasury_prov = { variable = crime_cost } }
			# Scripted Effects to reduce crime and take money
			decrease_crime_lvl_medium = yes
			add_province_modifier = { name = "anticrime_campaign" duration = 1825 }
		}
	}
	tooltip = "criminal_rating_tt"
}
