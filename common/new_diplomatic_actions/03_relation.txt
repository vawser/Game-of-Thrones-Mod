bribe_for_support = {
	category = relation

	alert_index = 39
	alert_tooltip = bribe_for_support_alert_tooltip

	require_acceptance = no

	is_visible = {
		OR = { # Only used by the two nations in question
			has_country_flag = petition_for_great_house_subject
			has_country_flag = petition_for_great_house_overlord
		}
		if = { # Only when specific incident is active
			limit = { active_imperial_incident = any }
			active_imperial_incident = incident_change_great_house
		}
		else = { always = no }
		# Can't use on overlord obviously
		FROM = { NOT = { overlord_of = ROOT } ai = yes }
	}
	is_allowed = {
		NOT = {
			reverse_has_opinion_modifier = {
				who = FROM
				modifier = bribed_for_support
			}
		}
		monthly_income = FROM
	}
	on_accept = {
		set_income_variable_with_tag = {
			income_var = briber_income
			percent_amount = 1.0
			tag = FROM
		}
		transfer_treasury = {
			receiver = FROM
			variable = briber_income
		}
		FROM = {
			add_opinion = {
				who = ROOT
				modifier = bribed_for_support
				months = 12
			}
			every_subject_country = {
				limit = { is_subject_of_type = feudatory }
				add_opinion = {
					who = ROOT
					modifier = bribed_for_support
					months = 12
				}
			}
		}
	}

	ai_will_do = {
		always = no
	}
}

establish_neutrality_pact = {
	category = relation

	alert_index = 39
	alert_tooltip = bribe_for_support_alert_tooltip

	require_acceptance = yes
	
	is_visible = {
		ai = no
		OR = {
			# Night's Watch with the wildlings
			AND = {
				tag = NGT
				FROM = { capital_scope = { region = region_beyond_the_wall } }
			}
			# Wildlings with the Night's Watch and other wildlings
			AND = {
				capital_scope = { region = region_beyond_the_wall }
				OR = {
					FROM = { tag = NGT }
					FROM = { capital_scope = { region = region_beyond_the_wall } }
				}
			}
		}
	}
	is_allowed = {
		if = {
			limit = { has_country_flag = established_neutrality_pact }
			custom_trigger_tooltip = {
				tooltip = "not_created_pact_in_last_year_tt"
				OR = {
					NOT = { has_country_flag = established_neutrality_pact }
					had_country_flag = {
						flag = established_neutrality_pact
						days = 365
					}
				}
			}
		}
		if = { # Not have a Neutrality Pact
			limit = { has_neutrality_pact = yes }
			custom_trigger_tooltip = {
				tooltip = "neutality_pact_active_tt"
				NOT = { has_neutrality_pact = yes }
			}
		}
		if = { # Is not King of SK
			limit = { FROM = { is_emperor = yes } }
			custom_trigger_tooltip = {
				tooltip = "no_pacts_with_sk_king_tt"
				NOT = { FROM = { is_emperor = yes } }
			}
		}
		if = { # Is not a subject
			limit = { FROM = { is_subject_of = ROOT } }
			custom_trigger_tooltip = {
				tooltip = "no_pacts_with_subjects_tt"
				NOT = { FROM = { is_subject_of = ROOT } }
			}
		}
	}
	on_accept = {
		add_neutrality_pact = yes
	}

	ai_acceptance = {
		# Base AI weight -30
		add_entry = {
			name = AIWEIGHT_BASE
			export_to_variable = {
				variable_name = ai_value
				value = -30
			}
		}
		# If the countries are neighbors, add 10 points
		add_entry = {
			name = NEIGHBOR_COUNTRY_STARTS
			limit = {
				is_neighbor_of = FROM
			}
			export_to_variable = {
				variable_name = ai_value
				value = 10
			}
		}
		# If the countries have a historical friendship, add 25 points
		add_entry = {
			name = historical_friend
			limit = {
				OR = {
					historical_friend_with = FROM
					FROM = { historical_friend_with = ROOT }
				}
			}
			export_to_variable = {
				variable_name = ai_value
				value = 25
			}
		}
		# If one country is threatened by the other, add 20 points
		add_entry = {
			name = attitude_threatened
			limit = {
				FROM = { is_threat = ROOT }
			}
			export_to_variable = {
				variable_name = ai_value
				value = 25
			}
		}
		# If the countries have a good diplomatic reputation, add 6 points per rep
		add_entry = {
			name = diplomatic_reputation
			limit = { }
			export_to_variable = {
				variable_name = ai_value
				value = trigger_value:diplomatic_reputation
			}
			#1 rep = 6
			multiply_variable = {
				which = ai_value
				value = 6
			}
		}
		# Add/reduce based on trust, baseline 50 trust and +/-6 points per 10 trust
		add_entry = {
			name = TRUST
			export_to_variable = {
				variable_name = ai_value
				value = trust
				who = FROM
				with = ROOT
			}			
			subtract_variable = { 
				which = ai_value
				value = 50
			}
			#10 trust = +-6 ai points
			multiply_variable = {
				which = ai_value
				value = 0.6
			}
		}
		# Add 0.1 points per opinion
		add_entry = {
			name = OPINION
			export_to_variable = {
				variable_name = ai_value
				value = opinion
				who = FROM
				with = ROOT
			}
			multiply_variable = {
				which = ai_value 
				value = 0.1
			}
		}
	}

	ai_will_do = {
		always = no
	}
}

invite_to_faction = {
	category = relation

	alert_index = 39
	alert_tooltip = bribe_for_support_alert_tooltip

	require_acceptance = yes
	
	is_visible = {
		ai = no
		overlord = { overlord_of = FROM }
	}
	is_allowed = {
		NOT = {
			OR = {
				is_rival = FROM
				is_enemy = FROM
			}
		}
	}
	on_accept = {
		FROM = { set_country_flag = AbdicationFaction@overlord }
	}

	ai_acceptance = {
		# Base AI weight -40
		add_entry = {
			name = AIWEIGHT_BASE
			export_to_variable = {
				variable_name = ai_value
				value = -40
			}
		}
		# If the countries have a good diplomatic reputation, add 5 points per rep
		add_entry = {
			name = diplomatic_reputation
			limit = { }
			export_to_variable = {
				variable_name = ai_value
				value = trigger_value:diplomatic_reputation
			}
			#1 rep = 5
			multiply_variable = {
				which = ai_value
				value = 5
			}
		}
		# Add 0.2 points per opinion
		add_entry = {
			name = OPINION
			export_to_variable = {
				variable_name = ai_value
				value = opinion
				who = FROM
				with = ROOT
			}
			multiply_variable = {
				which = ai_value 
				value = 0.2
			}
		}
		# Add 0.5 points per negative opinion with overlord
		add_entry = {
			name = OPINIONOVERLORD
			limit = {
				FROM = {
					overlord = {
						NOT = {
							reverse_has_opinion = {
								who = PREV
								value = 0
							}
						}
					}
				}
			}
			export_to_variable = {
				variable_name = ai_value
				value = opinion
				who = FROM
				with = overlord
			}
			multiply_variable = {
				which = ai_value
				value = -0.5
			}
		}
		# Add -25 points flat if loyal to overlord
		add_entry = {
			name = LOYAL
			limit = {
				overlord = {
					FROM = {
						ai_attitude = {
							who = PREV
							attitude = attitude_loyal
						}
					}
				}
			}
			export_to_variable = {
				variable_name = ai_value
				value = 1
			}
			multiply_variable = {
				which = ai_value
				value = -25
			}
		}
		# Add 50 points flat if disloyal to overlord
		add_entry = {
			name = DISLOYAL
			limit = {
				overlord = {
					FROM = {
						ai_attitude = {
							who = PREV
							attitude = attitude_disloyal
						}
					}
				}
			}
			export_to_variable = {
				variable_name = ai_value
				value = 1
			}
			multiply_variable = {
				which = ai_value
				value = 50
			}
		}
		# Add -1000 points flat if historical friends with overlord
		add_entry = {
			name = HISFRIENDS
			limit = {
				overlord = {
					FROM = { historical_friend_with = PREV }
				}
			}
			export_to_variable = {
				variable_name = ai_value
				value = -1000
			}
		}
	}

	ai_will_do = {
		always = no
	}
}
