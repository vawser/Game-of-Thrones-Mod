transfer_manpower = {
	hidden_effect = {
		while = {
			limit = {
				check_variable = {
					which = $variable$
					value = 1
				}
			}
			$receiver$ = {
				add_manpower = 0.001
			}
			add_manpower = -0.001
			subtract_variable = {
				which = $variable$
				value = 1
			}
		}
	}
	[[custom_tooltip]
		custom_tooltip = $custom_tooltip$
	]
}

# Criminal Rating Mechanic

# type : bandits, street_gangs, syndicates
set_criminal_activity = {
	custom_tooltip = set_criminal_activity_$type$
	increase_crime_lvl_large = yes
	set_province_flag = criminal_activity_$type$
	add_province_modifier = {
		name = active_$type$
		duration = -1
		hidden = yes
	}
}

clear_criminal_activity = {
	custom_tooltip = clr_criminal_activity_$type$
	decrease_crime_lvl_large = yes
	clr_province_flag = criminal_activity_$type$
	remove_province_modifier = active_$type$
}

# Strictly use when increasing crime lvl
# to prevent exceeding max (100)
increase_crime_lvl = {
	tooltip = { change_variable = { which = crime_lvl value = $amount$ } }
	for = {
		amount = $amount$
		effect = "
			if = {
				limit = { NOT = { has_max_crime_lvl = yes } }
				change_variable = {
					which = crime_lvl
					value = 1
				}
			}
		"
	}
	set_anti_crime_cost = yes
}

# Strictly use when reducing crime lvl
# to prevent negative values
reduce_crime_lvl = {
	tooltip = { change_variable = { which = crime_lvl value = -$amount$ } }
	for = {
		amount = $amount$
		effect = "
			if = {
				limit = { has_any_crime_lvl = yes }
				change_variable = {
					which = crime_lvl
					value = -1
				}
			}
		"
	}
	set_anti_crime_cost = yes
}

# Will set the cost to reduce crime lvl
set_anti_crime_cost = {
	hidden_effect = {
		# Get base development value
		export_to_variable = {
			which = crime_cost
			value = development
		}
		# Cost scales with crime lvl
		multiply_variable = {
			which = crime_cost
			which = crime_lvl
		}
		multiply_variable = {
			which = crime_cost
			value = 0.25
		}
		# Get value for halved cost
		set_variable = {
			which = halved_crime_cost
			which = crime_cost
		}
		multiply_variable = {
			which = halved_crime_cost
			value = 0.5
		}
	}
}

check_for_watch_commanders_effect = {
	if = {
		limit = { tag = NGT }
		if = {
			limit = { has_no_watch_commander = { province_id = 16 } castle_not_abandoned = { province_id = 16 } }
			NGT = { set_watch_commander = { province_id = 16 option = 0 } }
		}
		if = {
			limit = { has_no_watch_commander = { province_id = 1580 } castle_not_abandoned = { province_id = 1580 } }
			NGT = { set_watch_commander = { province_id = 1580 option = 1 } }
		}
		if = {
			limit = { has_no_watch_commander = { province_id = 1584 } castle_not_abandoned = { province_id = 1584 } }
			NGT = { set_watch_commander = { province_id = 1584 option = 2 } }
		}
		if = {
			limit = { has_no_watch_commander = { province_id = 18 } castle_not_abandoned = { province_id = 18 } }
			NGT = { set_watch_commander = { province_id = 18 option = 3 } }
		}
	}
}

calculate_royal_fleet_size_effect = {
	if = {
		limit = { tag = CRW }
		CRW = {
			export_to_variable = {
				variable_name = CrownlandsFleetSize
				value = trigger_value:navy_size
			}
			export_to_variable = {
				variable_name = VelaryonFleetSize
				value = trigger_value:navy_size
				who = A43
			}
			set_variable = {
				which = RoyalFleetSize
				value = 0
			}
			change_variable = {
				which = RoyalFleetSize
				which = CrownlandsFleetSize
			}
			change_variable = {
				which = RoyalFleetSize
				which = VelaryonFleetSize
			}
		}
	}
}
